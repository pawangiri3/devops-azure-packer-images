# 1. Create resource group first
az group create -n Custom-image-rg -l centralindia

# 2. Verify credentials
az account show

# 3. Run with proper location format
packer build `
  -var "client_id=$env:ARM_CLIENT_ID" `
  -var "client_secret=$env:ARM_CLIENT_SECRET" `
  -var "subscription_id=$env:ARM_SUBSCRIPTION_ID" `
  -var "tenant_id=$env:ARM_TENANT_ID" `
  -var "location=eastasia" `
  -var "image_version=1.0.1" `
  .
